def run(blueprint):
    """Run Formulator with the provided form blueprint."""
    form_blueprint = load_blueprint(blueprint)
    app = Formulator(form_blueprint)
    context = app.run()
    
    # Print context results as YAML
    print(yaml.dump(context))



if __name__ == '__main__':
    run()
